```tla+
---------------------------- MODULE MutualExclusion ----------------------------
EXTENDS Integers, TLAPS

CONSTANTS Proc

ASSUME ProcAssumption == Proc = {1, 2}

VARIABLES lock, pc

vars == <<lock, pc>>

Labels == {"l0", "l1", "cs", "l2"}

TypeOK == /\ lock \in {0, 1}
          /\ pc \in [Proc -> Labels]

Init == /\ lock = 1
        /\ pc = [p \in Proc |-> "l0"]

(* Non-critical section - no-op, move to l1 *)
l0(p) == /\ pc[p] = "l0"
         /\ pc' = [pc EXCEPT ![p] = "l1"]
         /\ lock' = lock

(* Acquire lock - wait until free, then atomically acquire *)
l1(p) == /\ pc[p] = "l1"
         /\ lock = 1
         /\ lock' = 0
         /\ pc' = [pc EXCEPT ![p] = "cs"]

(* Critical section - no-op, move to l2 *)
cs(p) == /\ pc[p] = "cs"
         /\ pc' = [pc EXCEPT ![p] = "l2"]
         /\ lock' = lock

(* Release lock - set lock back to 1 *)
l2(p) == /\ pc[p] = "l2"
         /\ lock' = 1
         /\ pc' = [pc EXCEPT ![p] = "l0"]

Step(p) == l0(p) \/ l1(p) \/ cs(p) \/ l2(p)

Next == \E p \in Proc : Step(p)

Spec == Init /\ [][Next]_vars

(* Mutual exclusion invariant *)
LockInv == /\ \A p, q \in Proc : (p # q) => ~(pc[p] \in {"cs", "l2"} /\ pc[q] \in {"cs", "l2"})
           /\ \A p \in Proc : pc[p] \in {"cs", "l2"} => lock = 0

MutualExclusion == \A p, q \in Proc : (p # q) => ~(pc[p] = "cs" /\ pc[q] = "cs")

-----------------------------------------------------------------------------
(* TLAPS Proofs *)

THEOREM TypeCorrect == Spec => []TypeOK
<1>1. Init => TypeOK
  BY ProcAssumption DEF Init, TypeOK, Labels
<1>2. TypeOK /\ [Next]_vars => TypeOK'
  <2> SUFFICES ASSUME TypeOK,
                      [Next]_vars
               PROVE  TypeOK'
    OBVIOUS
  <2>1. CASE Next
    <3> SUFFICES ASSUME NEW p \in Proc,
                        Step(p)
                 PROVE  TypeOK'
      BY <2>1 DEF Next
    <3>1. CASE l0(p)
      BY <3>1, ProcAssumption DEF l0, TypeOK, Labels
    <3>2. CASE l1(p)
      BY <3>2, ProcAssumption DEF l1, TypeOK, Labels
    <3>3. CASE cs(p)
      BY <3>3, ProcAssumption DEF cs, TypeOK, Labels
    <3>4. CASE l2(p)
      BY <3>4, ProcAssumption DEF l2, TypeOK, Labels
    <3>5. QED
      BY <3>1, <3>2, <3>3, <3>4 DEF Step
  <2>2. CASE UNCHANGED vars
    BY <2>2 DEF vars, TypeOK
  <2>3. QED
    BY <2>1, <2>2
<1>3. QED
  BY <1>1, <1>2, PTL DEF Spec

THEOREM LockInvariant == Spec => []LockInv
<1>1. Init => LockInv
  BY ProcAssumption DEF Init, LockInv
<1>2. TypeOK /\ LockInv /\ [Next]_vars => LockInv'
  <2> SUFFICES ASSUME TypeOK,
                      LockInv,
                      [Next]_vars
               PROVE  LockInv'
    OBVIOUS
  <2>1. CASE Next
    <3> SUFFICES ASSUME NEW p \in Proc,
                        Step(p)
                 PROVE  LockInv'
      BY <2>1 DEF Next
    <3>1. CASE l0(p)
      BY <3>1, ProcAssumption DEF l0, LockInv, TypeOK, Labels
    <3>2. CASE l1(p)
      BY <3>2, ProcAssumption DEF l1, LockInv, TypeOK, Labels
    <3>3. CASE cs(p)
      BY <3>3, ProcAssumption DEF cs, LockInv, TypeOK, Labels
    <3>4. CASE l2(p)
      BY <3>4, ProcAssumption DEF l2, LockInv, TypeOK, Labels
    <3>5. QED
      BY <3>1, <3>2, <3>3, <3>4 DEF Step
  <2>2. CASE UNCHANGED vars
    BY <2>2 DEF vars, LockInv
  <2>3. QED
    BY <2>1, <2>2
<1>3. QED
  BY <1>1, <1>2, TypeCorrect, PTL DEF Spec

THEOREM MutualExclusionTheorem == Spec => []MutualExclusion
<1>1. LockInv => MutualExclusion
  BY ProcAssumption DEF LockInv, MutualExclusion
<1>2. QED
  BY <1>1, LockInvariant, PTL

=============================================================================
```
