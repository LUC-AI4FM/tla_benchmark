        setx1: x := self;
        checky1: if (y /= 0) {
            backoff1: b[self] := FALSE;
            waity1: await y = 0;
            goto start1;
        };
        sety1: y := self;
        checkx1: if (x /= self) {
            clearb1: b[self] := FALSE;
            waitb1: j := 1;
            loopb1: while (j <= N) {
                checkb1: if (j /= self) {
                    awaitb1: await ~b[j];
                };
                incj1: j := j + 1;
            };
            checky2_1: if (y /= self) {
                failed1: failed := TRUE;
            };
        };
        enter1: if (~failed) {
            cs1: skip;
        };
        exit1: y := 0;
               b[self] := FALSE;
        reset1: failed := FALSE;
               goto start1;
    }
    
    fair process (P2 \in Proc2)
    variables j2 = 1, failed2 = FALSE;
    {
        start2: b[self] := TRUE;
        setx2: x := self;
        checky2: if (y /= 0) {
            backoff2: b[self] := FALSE;
            waity2: await y = 0;
            goto start2;
        };
        sety2: y := self;
        checkx2: if (x /= self) {
            clearb2: b[self] := FALSE;
            waitb2: j2 := 1;
            loopb2: while (j2 <= N) {
                checkb2: if (j2 /= self) {
                    awaitb2: await ~b[j2];
                };
                incj2: j2 := j2 + 1;
            };
            checky2_2: if (y /= self) {
                failed2_2: failed2 := TRUE;
            };
        };
        enter2: if (~failed2) {
            cs2: skip;
        };
        exit2: y := 0;
               b[self] := FALSE;
        reset2: failed2 := FALSE;
               goto start2;
    }
}

ProcSet == Proc1 \cup Proc2
