------------------------------- MODULE simple -------------------------------
EXTENDS Naturals, TLAPS

CONSTANT N

ASSUME NAssumption == N \in Nat \ {0}

(*
--algorithm Simple {
    variables x = [i \in 1..N |-> 0], y = [i \in 1..N |-> 0];
    process (proc \in 1..N) {
        a: x[self] := 1;
        b: y[self] := x[IF self = 1 THEN N ELSE self - 1];
    }
}
*)

\* BEGIN TRANSLATION
VARIABLES x, y, pc

vars == << x, y, pc >>

ProcSet == (1..N)

Init == (* Global variables *)
        /\ x = [i \in 1..N |-> 0]
        /\ y = [i \in 1..N |-> 0]
        /\ pc = [self \in ProcSet |-> "a"]

a(self) == /\ pc[self] = "a"
           /\ x' = [x EXCEPT ![self] = 1]
           /\ pc' = [pc EXCEPT ![self] = "b"]
           /\ y' = y

b(self) == /\ pc[self] = "b"
           /\ y' = [y EXCEPT ![self] = x[IF self = 1 THEN N ELSE self - 1]]
           /\ pc' = [pc EXCEPT ![self] = "Done"]
           /\ x' = x

proc(self) == a(self) \/ b(self)

Next == (\E self \in 1..N: proc(self))
            \/ (* Disjunct to prevent deadlock on termination *)
               ((\A self \in ProcSet: pc[self] = "Done") /\ UNCHANGED vars)

Spec == Init /\ [][Next]_vars /\ WF_vars(Next)

Termination == <>(\A self \in ProcSet: pc[self] = "Done")

\* END TRANSLATION

TypeOK == /\ x \in [1..N -> {0, 1}]
          /\ y \in [1..N -> {0, 1}]
          /\ pc \in [1..N -> {"a", "b", "Done"}]

PCorrect == (\A i \in 1..N: pc[i] = "Done") => (\E i \in 1..N: y[i] = 1)

Inv == /\ TypeOK
       /\ \A i \in 1..N: (pc[i] \in {"b", "Done"}) => (x[i] = 1)
       /\ \A i \in 1..N: (pc[i] = "Done") => 
            (y[i] = x[IF i = 1 THEN N ELSE i - 1])
       /\ ((\A i \in 1..N: pc[i] = "Done") => (\E i \in 1..N: y[i] = 1))

LEMMA InvImpliesPCorrect == Inv => PCorrect
BY DEF Inv, PCorrect, TypeOK

LEMMA InitImpliesInv == Init => Inv
<1>1. Init => TypeOK
  BY NAssumption DEF Init, TypeOK, ProcSet
<1>2. Init => \A i \in 1..N: (pc[i] \in {"b", "Done"}) => (x[i] = 1)
  BY DEF Init, ProcSet
<1>3. Init => \A i \in 1..N: (pc[i] = "Done") => 
        (y[i] = x[IF i = 1 THEN N ELSE i - 1])
  BY DEF Init, ProcSet
<1>4. Init => ((\A i \in 1..N: pc[i] = "Done") => (\E i \in 1..N: y[i] = 1))
  BY DEF Init, ProcSet
<1>5. QED
  BY <1>1, <1>2, <1>3, <1>4 DEF Inv

LEMMA InvPreserved == Inv /\ [Next]_vars => Inv'
<1> SUFFICES ASSUME Inv, [Next]_vars
             PROVE  Inv'
  OBVIOUS
<1>1. CASE \E self \in 1..N: a(self)
  BY <1>1, NAssumption DEF Inv, TypeOK, a, ProcSet
<1>2. CASE \E self \in 1..N: b(self)
  BY <1>2, NAssumption DEF Inv, TypeOK, b, ProcSet
<1>3. CASE (\A self \in ProcSet: pc[self] = "Done") /\ UNCHANGED vars
  BY <1>3 DEF Inv, TypeOK, vars, ProcSet
<1>4. CASE UNCHANGED vars
  BY <1>4 DEF Inv, TypeOK, vars
<1>5. QED
  BY <1>1, <1>2, <1>3, <1>4 DEF Next, proc

THEOREM Correctness == Spec => []PCorrect
<1>1. Init => Inv
  BY InitImpliesInv
<1>2. Inv /\ [Next]_vars => Inv'
  BY InvPreserved
<1>3. Inv => PCorrect
  BY InvImpliesPCorrect
<1>4. QED
  BY <1>1, <1>2, <1>3, PTL DEF Spec

THEOREM CorrectnessV2 == Spec => []PCorrect
<1>1. Init /\ [][Next]_vars => []Inv
  <2>1. Init => Inv
    BY InitImpliesInv
  <2>2. Inv /\ [Next]_vars => Inv'
    BY InvPreserved
  <2>3. QED
    BY <2>1, <2>2, PTL
<1>2. []Inv => []PCorrect
  BY InvImpliesPCorrect, PTL
<1>3. QED
  BY <1>1, <1>2, PTL DEF Spec
