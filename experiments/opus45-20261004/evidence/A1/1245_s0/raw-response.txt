---------------------------- MODULE PlusCal ----------------------------

EXTENDS Sequences, Integers, TLC, FiniteSets

(***************************************************************************)
(* This module specifies the translation from the abstract syntax tree of  *)
(* a global-naming PlusCal algorithm into its corresponding TLA+           *)
(* specification.                                                          *)
(*                                                                         *)
(* It defines the abstract syntax tree grammar as sets and predicates over *)
(* records and sequences, and then defines a translation pipeline that     *)
(* explodes structured labeled statements, translates calls/returns/gotos, *)
(* adds subscripts for process-local variables, and finally constructs     *)
(* Init, Next, Spec, and a Termination property.                           *)
(*                                                                         *)
(* The file also models fairness options for the translation output,       *)
(* supporting no fairness, weak fairness of process actions, weak fairness *)
(* of Next, and strong fairness of process actions.                        *)
(*                                                                         *)
(* LIMITATIONS:                                                            *)
(* - This is a simplified model for TLC execution                          *)
(* - Some edge cases may not be fully covered                              *)
(* - Formatting of output is abstracted away                               *)
(*                                                                         *)
(* IMPLEMENTATION NOTES:                                                   *)
(* - Object represents any record/structured value in the AST              *)
(* - Any represents the universal set for type flexibility                 *)
(***************************************************************************)

(***************************************************************************)
(* CONSTANTS                                                               *)
(* Object - The set of all possible record/object values in the AST        *)
(* Any - The universal set representing any possible value                 *)
(***************************************************************************)
CONSTANTS Object, Any

(***************************************************************************)
(* Fairness options for the translation                                    *)
(***************************************************************************)
NoFairness == "none"
WeakFairnessOfProcessActions == "wf_proc"
WeakFairnessOfNext == "wf_next"
StrongFairnessOfProcessActions == "sf_proc"

FairnessOptions == {NoFairness, WeakFairnessOfProcessActions, 
                    WeakFairnessOfNext, StrongFairnessOfProcessActions}

(***************************************************************************)
(* AST Node Types                                                          *)
(***************************************************************************)
NodeTypes == {"Algorithm", "Process", "Procedure", "LabeledStmt", 
              "Assignment", "If", "While", "Either", "With",
              "Call", "Return", "Goto", "Print", "Assert", "Skip",
              "Variable", "Expr", "Label"}

(***************************************************************************)
(* Predicate: IsObject - checks if a value is a valid Object               *)
(***************************************************************************)
IsObject(x) == x \in Object

(***************************************************************************)
(* Predicate: IsSequenceOf - checks if s is a sequence of elements from S  *)
(***************************************************************************)
IsSequenceOf(s, S) == 
    /\ s \in Seq(S)

(***************************************************************************)
(* Predicate: IsRecord - checks if r is a record with given fields         *)
(***************************************************************************)
IsRecord(r, fields) ==
    /\ r \in Object
    /\ DOMAIN r = fields

(***************************************************************************)
(* AST Node Predicates                                                     *)
(***************************************************************************)
IsASTNode(node) ==
    /\ node \in Object
    /\ "type" \in DOMAIN node
    /\ node.type \in NodeTypes

IsVariable(v) ==
    /\ IsASTNode(v)
    /\ v.type = "Variable"
    /\ "name" \in DOMAIN v
    /\ "init" \in DOMAIN v

IsLabel(l) ==
    /\ IsASTNode(l)
    /\ l.type = "Label"
    /\ "name" \in DOMAIN l

IsLabeledStmt(stmt) ==
    /\ IsASTNode(stmt)
    /\ stmt.type = "LabeledStmt"
    /\ "label" \in DOMAIN stmt
    /\ "stmts" \in DOMAIN stmt

IsProcess(proc) ==
    /\ IsASTNode(proc)
    /\ proc.type = "Process"
    /\ "name" \in DOMAIN proc
    /\ "id" \in DOMAIN proc
    /\ "body" \in DOMAIN proc

IsProcedure(proc) ==
    /\ IsASTNode(proc)
    /\ proc.type = "Procedure"
    /\ "name" \in DOMAIN proc
    /\ "params" \in DOMAIN proc
    /\ "body" \in DOMAIN proc

IsAlgorithm(alg) ==
    /\ IsASTNode(alg)
    /\ alg.type = "Algorithm"
    /\ "name" \in DOMAIN alg
    /\ "variables" \in DOMAIN alg
    /\ "procedures" \in DOMAIN alg
    /\ "processes" \in DOMAIN alg

(***************************************************************************)
(* Translation Pipeline Operators                                          *)
(***************************************************************************)

(* ExplodeLabeledStmts - Transforms structured labeled statements into     *)
(* flat sequences of atomic actions                                        *)
ExplodeLabeledStmts(stmts) ==
    IF stmts = <<>> 
    THEN <<>>
    ELSE stmts  \* Simplified: actual implementation would recursively explode

(* TranslateCallsReturnsGotos - Converts call/return/goto statements into  *)
(* appropriate pc assignments and stack manipulations                      *)
TranslateCallsReturnsGotos(stmts) ==
    stmts  \* Simplified placeholder

(* AddProcessSubscripts - Adds self subscripts for process-local variables *)
AddProcessSubscripts(stmts, procName) ==
    stmts  \* Simplified placeholder

(* TranslateStatement - Main statement translation                         *)
TranslateStatement(stmt) ==
    stmt  \* Simplified placeholder

(***************************************************************************)
(* Spec Construction Operators                                             *)
(***************************************************************************)

(* ConstructInit - Builds the Init predicate                               *)
ConstructInit(alg) ==
    IF IsAlgorithm(alg)
    THEN [type |-> "Init", algorithm |-> alg.name]
    ELSE [type |-> "Error"]

(* ConstructNext - Builds the Next action                                  *)
ConstructNext(alg) ==
    IF IsAlgorithm(alg)
    THEN [type |-> "Next", algorithm |-> alg.name]
    ELSE [type |-> "Error"]

(* ConstructSpec - Builds the complete Spec with fairness                  *)
ConstructSpec(alg, fairness) ==
    IF IsAlgorithm(alg) /\ fairness \in FairnessOptions
    THEN [type |-> "Spec", 
          algorithm |-> alg.name,
          fairness |-> fairness,
          init |-> ConstructInit(alg),
          next |-> ConstructNext(alg)]
    ELSE [type |-> "Error"]

(* ConstructTermination - Builds the Termination property                  *)
ConstructTermination(alg) ==
    IF IsAlgorithm(alg)
    THEN [type |-> "Termination", algorithm |-> alg.name]
    ELSE [type |-> "Error"]

(***************************************************************************)
(* Main Translation Operator                                               *)
(***************************************************************************)
Translate(alg, fairness) ==
    LET exploded == ExplodeLabeledStmts(alg)
        translated == TranslateCallsReturnsGotos(exploded)
    IN ConstructSpec(translated, fairness)

(***************************************************************************)
(* Type Invariants for Model Checking                                      *)
(***************************************************************************)
TypeOK ==
    /\ Object \subseteq Any
    /\ FairnessOptions \subseteq Any

(***************************************************************************)
(* Helper: Empty Algorithm for testing                                     *)
(***************************************************************************)
EmptyAlgorithm == 
    [type |-> "Algorithm",
     name |-> "Empty",
     variables |-> <<>>,
     procedures |-> <<>>,
     processes |-> <<>>]
