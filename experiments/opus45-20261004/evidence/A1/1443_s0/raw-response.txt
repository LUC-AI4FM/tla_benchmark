---------------------------- MODULE Specification ----------------------------

EXTENDS Integers, TLC

VARIABLE x

Init == x = 0

Next == x' = (x + 1) % 3

IsOne == x = 1

Done == x = 2

Wrap == x = 2 /\ x' = 0

Spec == Init /\ [][Next]_x

CheckCoverage ==
    LET isOneCnt == TLCGet("stats").coverage["IsOne"]
        doneCnt == TLCGet("stats").coverage["Done"]
        wrapCnt == TLCGet("stats").coverage["Wrap"]
    IN isOneCnt = 1 /\ doneCnt = 1 /\ wrapCnt = 1

=============================================================================
