---------------------------- MODULE BarrierSync ----------------------------

EXTENDS Naturals, FiniteSets

CONSTANTS N

ASSUME N \in Nat /\ N > 0

VARIABLES pc

Procs == 1..N

TypeOK == pc \in [Procs -> {"b0", "b1"}]

Init == pc = [p \in Procs |-> "b0"]

Enter(p) == 
    /\ pc[p] = "b0"
    /\ pc' = [pc EXCEPT ![p] = "b1"]

Reset == 
    /\ \A p \in Procs : pc[p] = "b1"
    /\ pc' = [p \in Procs |-> "b0"]

Next == 
    \/ \E p \in Procs : Enter(p)
    \/ Reset

Spec == Init /\ [][Next]_pc /\ WF_pc(Next)

AllAtB1 == \A p \in Procs : pc[p] = "b1"

SomeNotEntered == \E p \in Procs : pc[p] = "b0"

BarrierProperty == 
    [](\A p \in Procs : (pc[p] = "b1" /\ SomeNotEntered) => pc'[p] # "b0")

BarrierInvariant == 
    \A p, q \in Procs : 
        (pc[p] = "b0" /\ pc[q] = "b1") => 
            (pc[p] = "b0" \/ \A r \in Procs : pc[r] = "b1")

NoPartialReset == 
    ~(\E p, q \in Procs : pc[p] = "b0" /\ pc[q] = "b1" /\ 
        \E r \in Procs : pc[r] = "b1" /\ r # q)

SafetyInvariant == 
    \A p \in Procs : 
        pc[p] = "b0" => 
            (\A q \in Procs : pc[q] = "b0") \/ (\A q \in Procs : pc[q] = "b1")

Liveness == \A p \in Procs : (pc[p] = "b1") ~> (pc[p] = "b0")
