```tla
---- MODULE GCD ----
EXTENDS Integers

CONSTANTS InitialU, MaxV

ASSUME InitialU \in Nat /\ InitialU > 0
ASSUME MaxV \in Nat /\ MaxV > 0

VARIABLES u, v, pc

vars == <<u, v, pc>>

Init ==
    /\ u = InitialU
    /\ v \in 1..MaxV
    /\ pc = "loop"

Swap ==
    /\ pc = "loop"
    /\ u < v
    /\ u' = v
    /\ v' = u
    /\ pc' = "loop"

Subtract ==
    /\ pc = "loop"
    /\ u >= v
    /\ u # 0
    /\ u' = u - v
    /\ v' = v
    /\ pc' = "loop"

Terminate ==
    /\ pc = "loop"
    /\ u = 0
    /\ pc' = "Done"
    /\ u' = u
    /\ v' = v

Done ==
    /\ pc = "Done"
    /\ UNCHANGED vars

Next ==
    \/ Swap
    \/ Subtract
    \/ Terminate
    \/ Done

Spec == Init /\ [][Next]_vars

TypeInvariant ==
    /\ u \in Nat
    /\ v \in Nat
    /\ pc \in {"loop", "Done"}

PositiveV ==
    v > 0

GCDPreserved ==
    pc = "loop" => v > 0

Termination ==
    <>(pc = "Done")

====
```
