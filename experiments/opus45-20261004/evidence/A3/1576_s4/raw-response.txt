```tla
---------------------------- MODULE ConcurrentQueue ----------------------------
EXTENDS Naturals, Sequences

CONSTANTS Procs, Val, N

VARIABLES queue, rV, pc

vars == <<queue, rV, pc>>

Init ==
    /\ queue = <<>>
    /\ rV = [p \in Procs |-> "null"]
    /\ pc = [p \in Procs |-> "choose"]

EnqueueFront(self) ==
    /\ pc[self] = "choose"
    /\ Len(queue) < N
    /\ \E v \in Val:
        /\ queue' = <<v>> \o queue
        /\ rV' = [rV EXCEPT ![self] = "okay"]
    /\ pc' = [pc EXCEPT ![self] = "choose"]

EnqueueBack(self) ==
    /\ pc[self] = "choose"
    /\ Len(queue) < N
    /\ \E v \in Val:
        /\ queue' = Append(queue, v)
        /\ rV' = [rV EXCEPT ![self] = "okay"]
    /\ pc' = [pc EXCEPT ![self] = "choose"]

EnqueueFullFront(self) ==
    /\ pc[self] = "choose"
    /\ Len(queue) >= N
    /\ queue' = queue
    /\ rV' = [rV EXCEPT ![self] = "full"]
    /\ pc' = [pc EXCEPT ![self] = "choose"]

EnqueueFullBack(self) ==
    /\ pc[self] = "choose"
    /\ Len(queue) >= N
    /\ queue' = queue
    /\ rV' = [rV EXCEPT ![self] = "full"]
    /\ pc' = [pc EXCEPT ![self] = "choose"]

DequeueHead(self) ==
    /\ pc[self] = "choose"
    /\ Len(queue) > 0
    /\ rV' = [rV EXCEPT ![self] = Head(queue)]
    /\ queue' = Tail(queue)
    /\ pc' = [pc EXCEPT ![self] = "choose"]

DequeueTail(self) ==
    /\ pc[self] = "choose"
    /\ Len(queue) > 0
    /\ rV' = [rV EXCEPT ![self] = queue[Len(queue)]]
    /\ queue' = SubSeq(queue, 1, Len(queue) - 1)
    /\ pc' = [pc EXCEPT ![self] = "choose"]

DequeueEmptyHead(self) ==
    /\ pc[self] = "choose"
    /\ Len(queue) = 0
    /\ queue' = queue
    /\ rV' = [rV EXCEPT ![self] = "empty"]
    /\ pc' = [pc EXCEPT ![self] = "choose"]

DequeueEmptyTail(self) ==
    /\ pc[self] = "choose"
    /\ Len(queue) = 0
    /\ queue' = queue
    /\ rV' = [rV EXCEPT ![self] = "empty"]
    /\ pc' = [pc EXCEPT ![self] = "choose"]

P(self) ==
    \/ EnqueueFront(self)
    \/ EnqueueBack(self)
    \/ EnqueueFullFront(self)
    \/ EnqueueFullBack(self)
    \/ DequeueHead(self)
    \/ DequeueTail(self)
    \/ DequeueEmptyHead(self)
    \/ DequeueEmptyTail(self)

Next ==
    \E self \in Procs: P(self)

Fairness ==
    \A self \in Procs: WF_vars(P(self))

Spec ==
    /\ Init
    /\ [][Next]_vars
    /\ Fairness

StateConstraint ==
    Len(queue) <= N

TypeInvariant ==
    /\ queue \in Seq(Val)
    /\ Len(queue) <= N
    /\ pc \in [Procs -> {"choose"}]
    /\ rV \in [Procs -> Val \cup {"okay", "full", "empty", "null"}]

QueueBoundInvariant ==
    Len(queue) <= N
```
