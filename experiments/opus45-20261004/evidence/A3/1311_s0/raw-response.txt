-------------------------------- MODULE gcd --------------------------------
EXTENDS Integers

CONSTANTS ValRange

VARIABLES u, v, pc

vars == <<u, v, pc>>

Init ==
    /\ u = 24
    /\ v \in 1..50
    /\ pc = "loop"

Swap ==
    /\ pc = "loop"
    /\ u < v
    /\ u' = v
    /\ v' = u
    /\ pc' = "loop"

Subtract ==
    /\ pc = "loop"
    /\ u >= v
    /\ u # 0
    /\ u' = u - v
    /\ v' = v
    /\ pc' = "loop"

Terminate ==
    /\ pc = "loop"
    /\ u = 0
    /\ pc' = "Done"
    /\ u' = u
    /\ v' = v

Done ==
    /\ pc = "Done"
    /\ UNCHANGED vars

Next ==
    \/ Swap
    \/ Subtract
    \/ Terminate
    \/ Done

Spec == Init /\ [][Next]_vars

TypeInvariant ==
    /\ u \in Nat
    /\ v \in 1..50
    /\ pc \in {"loop", "Done"}

NonNegativeInvariant ==
    /\ u >= 0
    /\ v >= 0

VPositiveInvariant ==
    v > 0

GCDPreservation ==
    pc = "Done" => u = 0

=============================================================================
