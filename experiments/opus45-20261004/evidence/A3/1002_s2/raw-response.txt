```tla
---------------------------- MODULE BarrierSync ----------------------------

EXTENDS Integers, FiniteSets

CONSTANTS N

ASSUME N \in Nat /\ N > 0

VARIABLES pc

Procs == 1..N

TypeOK == pc \in [Procs -> {"b0", "b1"}]

Init == pc = [p \in Procs |-> "b0"]

Enter(p) == 
    /\ pc[p] = "b0"
    /\ pc' = [pc EXCEPT ![p] = "b1"]

Reset == 
    /\ \A p \in Procs : pc[p] = "b1"
    /\ pc' = [p \in Procs |-> "b0"]

Next == 
    \/ \E p \in Procs : Enter(p)
    \/ Reset

Spec == Init /\ [][Next]_pc /\ WF_pc(Next)

BarrierProperty == 
    \A p \in Procs : \A q \in Procs : 
        (pc[p] = "b0" /\ pc[q] = "b1") => (pc'[q] = "b1" \/ pc'[p] = "b1")

SafetyInvariant == 
    \A p \in Procs : \A q \in Procs :
        ~(pc[p] = "b0" /\ pc[q] = "b1" /\ 
          (\E r \in Procs : pc[r] = "b0" /\ r # q))
        => TRUE

NoLeaveWhileOthersOutside ==
    [][
        \A p \in Procs : \A q \in Procs :
            (pc[q] = "b0" /\ q # p) => ~(pc[p] = "b1" /\ pc'[p] = "b0" /\ pc'[q] = "b0")
    ]_pc

BarrierSafety == 
    [][\A p \in Procs : 
        (pc[p] = "b1" /\ pc'[p] = "b0") => 
        (\A q \in Procs : pc[q] = "b1")
    ]_pc

EventualReset == 
    \A p \in Procs : pc[p] = "b1" ~> pc[p] = "b0"

=============================================================================
```
