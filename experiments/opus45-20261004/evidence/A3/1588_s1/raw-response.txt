```tla
---------------------------- MODULE EuclidAlgorithm ----------------------------
EXTENDS Naturals, TLC

CONSTANT N

VARIABLES pc, u, v, v_ini

vars == <<pc, u, v, v_ini>>

Init ==
    /\ u = 24
    /\ v_ini \in 1..N
    /\ v = v_ini
    /\ pc = "start"

GCD(a, b) ==
    LET GCDSet == {d \in 1..a : (a % d = 0) /\ (b % d = 0)}
    IN IF a > 0 /\ b > 0 THEN CHOOSE x \in GCDSet : \A y \in GCDSet : x >= y ELSE 1

Next ==
    \/ /\ pc = "start"
       /\ u # v
       /\ IF u > v
          THEN /\ u' = u - v
               /\ v' = v
          ELSE /\ v' = v - u
               /\ u' = u
       /\ pc' = "start"
       /\ v_ini' = v_ini
    \/ /\ pc = "start"
       /\ u = v
       /\ pc' = "Done"
       /\ UNCHANGED <<u, v, v_ini>>

Spec == Init /\ [][Next]_vars /\ WF_vars(Next)

TypeOK ==
    /\ u \in Nat
    /\ v \in Nat
    /\ v_ini \in 1..N
    /\ pc \in {"start", "Done"}

PositiveValues ==
    /\ u > 0
    /\ v > 0

GCDInvariant ==
    pc = "Done" => u = GCD(24, v_ini)

Termination == <>(pc = "Done")

=============================================================================
```
