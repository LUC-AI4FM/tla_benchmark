```tla
---------------------------- MODULE mergesort ----------------------------
EXTENDS Integers, Sequences, FiniteSets, TLC

CONSTANTS ArrayLen, N

ASSUME ArrayLen \in Nat /\ ArrayLen >= 0
ASSUME N \in Nat /\ N >= 1

VARIABLES a, b, pc, stack, lo, hi, mid

vars == <<a, b, pc, stack, lo, hi, mid>>

Values == 1..N
Indexes == 1..ArrayLen

IsSorted(arr, from, to) ==
    \A i, j \in from..to : i < j => arr[i] <= arr[j]

ArrayIsSorted ==
    pc = "Done" => IsSorted(a, 1, ArrayLen)

Init ==
    /\ a \in [1..ArrayLen -> Values]
    /\ b = [i \in 1..ArrayLen |-> 1]
    /\ pc = IF ArrayLen <= 1 THEN "Done" ELSE "MergeSort"
    /\ stack = <<>>
    /\ lo = 1
    /\ hi = ArrayLen
    /\ mid = 1

MergeSort ==
    /\ pc = "MergeSort"
    /\ IF hi - lo < 1
       THEN /\ pc' = "Return"
            /\ UNCHANGED <<a, b, stack, lo, hi, mid>>
       ELSE /\ mid' = lo + ((hi - lo) \div 2)
            /\ pc' = "SortLeft"
            /\ UNCHANGED <<a, b, stack, lo, hi>>

SortLeft ==
    /\ pc = "SortLeft"
    /\ stack' = <<[lo |-> lo, hi |-> hi, mid |-> mid, ret |-> "SortRight"]>> \o stack
    /\ hi' = mid
    /\ pc' = "MergeSort"
    /\ UNCHANGED <<a, b, lo, mid>>

SortRight ==
    /\ pc = "SortRight"
    /\ stack' = <<[lo |-> lo, hi |-> hi, mid |-> mid, ret |-> "Merge"]>> \o stack
    /\ lo' = mid + 1
    /\ pc' = "MergeSort"
    /\ UNCHANGED <<a, b, hi, mid>>

Merge ==
    /\ pc = "Merge"
    /\ LET 
           left == lo
           right == mid + 1
           MergeLoop[state \in [l: Nat, r: Nat, k: Nat, buf: [1..ArrayLen -> Values]]] ==
               IF state.k > hi
               THEN state.buf
               ELSE IF state.l > mid
                    THEN MergeLoop[[l |-> state.l, r |-> state.r + 1, k |-> state.k + 1,
                                    buf |-> [state.buf EXCEPT ![state.k] = a[state.r]]]]
                    ELSE IF state.r > hi
                         THEN MergeLoop[[l |-> state.l + 1, r |-> state.r, k |-> state.k + 1,
                                         buf |-> [state.buf EXCEPT ![state.k] = a[state.l]]]]
                         ELSE IF a[state.l] <= a[state.r]
                              THEN MergeLoop[[l |-> state.l + 1, r |-> state.r, k |-> state.k + 1,
                                              buf |-> [state.buf EXCEPT ![state.k] = a[state.l]]]]
                              ELSE MergeLoop[[l |-> state.l, r |-> state.r + 1, k |-> state.k + 1,
                                              buf |-> [state.buf EXCEPT ![state.k] = a[state.r]]]]
           mergedBuf == MergeLoop[[l |-> left, r |-> right, k |-> lo, buf |-> b]]
           CopyBack[arr \in [1..ArrayLen -> Values], idx \in Nat] ==
               IF idx > hi
               THEN arr
               ELSE CopyBack[[arr EXCEPT ![idx] = mergedBuf[idx]], idx + 1]
       IN
           /\ b' = mergedBuf
           /\ a' = CopyBack[a, lo]
           /\ pc' = "Return"
           /\ UNCHANGED <<stack, lo, hi, mid>>

Return ==
    /\ pc = "Return"
    /\ IF stack = <<>>
       THEN /\ pc' = "Done"
            /\ UNCHANGED <<a, b, stack, lo, hi, mid>>
       ELSE /\ LET frame == Head(stack)
            IN /\ lo' = frame.lo
               /\ hi' = frame.hi
               /\ mid' = frame.mid
               /\ pc' = frame.ret
               /\ stack' = Tail(stack)
            /\ UNCHANGED <<a, b>>

Done ==
    /\ pc = "Done"
    /\ UNCHANGED vars

Next ==
    \/ MergeSort
    \/ SortLeft
    \/ SortRight
    \/ Merge
    \/ Return
    \/ Done

Spec == Init /\ [][Next]_vars /\ WF_vars(Next)

TypeInvariant ==
    /\ a \in [1..ArrayLen -> Values]
    /\ b \in [1..ArrayLen -> Values]
    /\ pc \in {"MergeSort", "SortLeft", "SortRight", "Merge", "Return", "Done"}
    /\ lo \in 1..ArrayLen
    /\ hi \in 1..ArrayLen
    /\ mid \in 1..ArrayLen

Safety == ArrayIsSorted

Termination == <>(pc = "Done")
```
