---------------------------- MODULE BarrierSync ----------------------------

EXTENDS Integers, FiniteSets

CONSTANTS N

ASSUME N \in Nat /\ N > 0

VARIABLES pc

Procs == 1..N

TypeOK == pc \in [Procs -> {"b0", "b1"}]

Init == pc = [p \in Procs |-> "b0"]

Enter(p) == 
    /\ pc[p] = "b0"
    /\ pc' = [pc EXCEPT ![p] = "b1"]

Reset == 
    /\ \A p \in Procs : pc[p] = "b1"
    /\ pc' = [p \in Procs |-> "b0"]

Next == 
    \/ \E p \in Procs : Enter(p)
    \/ Reset

Spec == Init /\ [][Next]_pc

FairSpec == Spec /\ WF_pc(Next)

AllInB1 == \A p \in Procs : pc[p] = "b1"

SomeInB0 == \E p \in Procs : pc[p] = "b0"

BarrierProperty == 
    \A p \in Procs : 
        (pc[p] = "b1" /\ SomeInB0) => pc'[p] = "b1"

BarrierInvariant == 
    \A p, q \in Procs : 
        (pc[p] = "b0" /\ pc[q] = "b1") => 
            \A r \in Procs : pc[r] # "b0" \/ r = p \/ pc[r] = "b1"

SafetyProperty == 
    [][(\E p \in Procs : pc[p] = "b1" /\ pc'[p] = "b0") => 
       (\A q \in Procs : pc[q] = "b1")]_pc

LivenessProperty == 
    \A p \in Procs : (pc[p] = "b1") ~> (pc[p] = "b0")

=============================================================================
