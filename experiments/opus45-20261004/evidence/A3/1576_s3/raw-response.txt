---------------------------- MODULE ConcurrentQueue ----------------------------
EXTENDS Integers, Sequences

CONSTANTS Procs, Val, N

VARIABLES queue, rV, pc

vars == <<queue, rV, pc>>

\* Possible return values
ReturnValues == {"okay", "full", "empty", "null"} \cup Val

\* Initial state
Init ==
    /\ queue = <<>>
    /\ rV = [p \in Procs |-> "null"]
    /\ pc = [p \in Procs |-> "choose"]

\* Process actions
Choose(self) ==
    /\ pc[self] = "choose"
    /\ \E op \in {"enqueue_front", "enqueue_back", "dequeue_head", "dequeue_tail"} :
        pc' = [pc EXCEPT ![self] = op]
    /\ UNCHANGED <<queue, rV>>

EnqueueFront(self) ==
    /\ pc[self] = "enqueue_front"
    /\ IF Len(queue) < N
       THEN \E v \in Val :
            /\ queue' = <<v>> \o queue
            /\ rV' = [rV EXCEPT ![self] = "okay"]
       ELSE 
            /\ rV' = [rV EXCEPT ![self] = "full"]
            /\ UNCHANGED queue
    /\ pc' = [pc EXCEPT ![self] = "choose"]

EnqueueBack(self) ==
    /\ pc[self] = "enqueue_back"
    /\ IF Len(queue) < N
       THEN \E v \in Val :
            /\ queue' = Append(queue, v)
            /\ rV' = [rV EXCEPT ![self] = "okay"]
       ELSE 
            /\ rV' = [rV EXCEPT ![self] = "full"]
            /\ UNCHANGED queue
    /\ pc' = [pc EXCEPT ![self] = "choose"]

DequeueHead(self) ==
    /\ pc[self] = "dequeue_head"
    /\ IF Len(queue) > 0
       THEN 
            /\ rV' = [rV EXCEPT ![self] = Head(queue)]
            /\ queue' = Tail(queue)
       ELSE 
            /\ rV' = [rV EXCEPT ![self] = "empty"]
            /\ UNCHANGED queue
    /\ pc' = [pc EXCEPT ![self] = "choose"]

DequeueTail(self) ==
    /\ pc[self] = "dequeue_tail"
    /\ IF Len(queue) > 0
       THEN 
            /\ rV' = [rV EXCEPT ![self] = queue[Len(queue)]]
            /\ queue' = SubSeq(queue, 1, Len(queue) - 1)
       ELSE 
            /\ rV' = [rV EXCEPT ![self] = "empty"]
            /\ UNCHANGED queue
    /\ pc' = [pc EXCEPT ![self] = "choose"]

\* Combined process action
P(self) ==
    \/ Choose(self)
    \/ EnqueueFront(self)
    \/ EnqueueBack(self)
    \/ DequeueHead(self)
    \/ DequeueTail(self)

\* Next state relation
Next ==
    \E self \in Procs : P(self)

\* Specification with weak fairness for each process
Spec ==
    /\ Init
    /\ [][Next]_vars
    /\ \A self \in Procs : WF_vars(P(self))

\* State constraint: queue length bounded by N
StateConstraint == Len(queue) <= N

\* Safety invariants
TypeInvariant ==
    /\ queue \in Seq(Val)
    /\ Len(queue) <= N
    /\ rV \in [Procs -> ReturnValues]
    /\ pc \in [Procs -> {"choose", "enqueue_front", "enqueue_back", "dequeue_head", "dequeue_tail"}]

QueueBoundedInvariant == Len(queue) <= N

\* Liveness properties
\* Each process eventually gets to make a choice
ProcessProgress == \A self \in Procs : []<>(pc[self] = "choose")
