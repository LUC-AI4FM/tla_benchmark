(* This module models a TLC issue involving an identifier becoming         *)

(* The purpose is to test the behavior when a constant-valued operator     *)
(* is substituted for a variable in an instantiated module.                *)

fizzbuzz == 42

INSTANCE x_unchanged WITH x <- fizzbuzz
